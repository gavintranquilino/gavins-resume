\DeclareResumeBullet{lidar-navigation/point-1}{%
  Developed \textbf{C++ ROS2} nodes to convert \textbf{LiDAR} data into a \textbf{2D costmap} for obstacle detection and perception.}

\DeclareResumeBullet{lidar-navigation/point-2}{%
  Generated a \textbf{world model} from costmap and odometry data to represent the current environment.}

\DeclareResumeBullet{lidar-navigation/point-3}{%
  Implemented \textbf{A* algorithm} to compute obstacle-aware paths through the mapped environment.}

\DeclareResumeBullet{lidar-navigation/point-4}{%
  Applied \textbf{Pure Pursuit} to follow planned paths for smooth differential drive navigation.}

\DeclareResumeBullet{lidar-navigation/point-5}{%
  \textbf{Dockerized} the system and integrated with \textbf{Gazebo} and \textbf{Foxglove} for simulation, debugging, and real-time visualization.}
